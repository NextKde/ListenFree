cmake_minimum_required(VERSION 3.28)
project(ListenFreeDependencyCheck LANGUAGES C CXX)
# Qmmp's build requires LinguistTools even though the app uses JSON translations.
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickDialogs2 Sql Network Concurrent DBus ShaderTools LinguistTools)
find_package(KF6WindowSystem REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
find_package(ICU REQUIRED COMPONENTS uc i18n)
find_path(UTF8CPP_INCLUDE NAMES utf8.h PATH_SUFFIXES utf8cpp REQUIRED)
if(LISTENFREE_BUILD_VIDEO)
    find_package(Qt6 6.10 REQUIRED COMPONENTS Multimedia)
endif()
if(LISTENFREE_BUILD_WEB_LOGIN)
    find_package(Qt6 6.10 REQUIRED COMPONENTS WebEngineWidgets)
endif()
get_target_property(qtpaths Qt6::qtpaths IMPORTED_LOCATION)
if(NOT qtpaths)
    get_target_property(configurations Qt6::qtpaths IMPORTED_CONFIGURATIONS)
    foreach(configuration IN LISTS configurations)
        get_target_property(qtpaths Qt6::qtpaths IMPORTED_LOCATION_${configuration})
        if(qtpaths)
            break()
        endif()
    endforeach()
endif()
if(NOT qtpaths)
    message(FATAL_ERROR "The selected Qt toolchain must provide qtpaths")
endif()
execute_process(COMMAND "${qtpaths}" --query QT_INSTALL_QML OUTPUT_VARIABLE qml_root OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${qtpaths}" --query QT_INSTALL_PLUGINS OUTPUT_VARIABLE plugin_root OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT EXISTS "${plugin_root}/sqldrivers/libqsqlite.so")
    message(FATAL_ERROR "Missing Qt SQLite runtime driver: ${plugin_root}/sqldrivers/libqsqlite.so")
endif()
set(modules QtCore QtQml QtQml/Models QtQml/WorkerScript QtQuick QtQuick/Window QtQuick/Layouts QtQuick/Controls QtQuick/Templates QtQuick/Dialogs QtQuick/Effects QtQuick/Shapes)
if(LISTENFREE_BUILD_VIDEO)
    list(APPEND modules QtMultimedia)
endif()
foreach(module IN LISTS modules)
    if(NOT EXISTS "${qml_root}/${module}/qmldir")
        message(FATAL_ERROR "Missing QML runtime module ${module} under ${qml_root}")
    endif()
endforeach()
